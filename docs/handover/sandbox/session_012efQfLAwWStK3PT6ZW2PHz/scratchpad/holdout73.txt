173:/**
174: * One write: checks the shared copy, reads the store, applies `fn` and publishes the result as one commit. `fn` may
175: * return an error to raise after the result is published (a refusal that burns must still be recorded).

247:export const registerAttempt = (a: HoldoutAuthority,
248:  req: { readonly index: number; readonly entries: readonly { readonly holdoutId: string; readonly universe: string; readonly configId: string }[]; readonly fromDay?: string },
249:  now: Date = new Date()): HoldoutStore =>
250:  mutate(a, `Holdout attempt ${req.index}: registered ${req.entries.map((e) => e.holdoutId).join(', ')}`, (s) => {
251:    if (s === null || s.plan === null) throw new RangeError('no holdout plan: set the plan before registering an attempt');
252:    if (req.entries.length === 0) throw new RangeError('an attempt needs at least one holdout');
253:    const plan = s.plan;
254:    const next = s.attempts.length + 1;
255:    if (req.index !== next) {
256:      let reg = s.registry;
257:      const why = `attempt ${req.index} asked, next is ${next}`;
258:      const repeated = s.attempts.find((x) => x.index === req.index);
259:      const ids = [...req.entries.map((e) => e.holdoutId), ...(repeated?.holdoutIds ?? [])];
260:      for (const id of ids) {
261:        const e = reg.entries.find((x) => x.holdoutId === id);
262:        if (e !== undefined && !e.burned) reg = burnHoldout(reg, id, 'reconfigured', why).registry;
263:      }
264:      return { store: { ...s, registry: reg }, value: s, error: new RangeError(`holdout attempt refused: ${why}`) };
265:    }
266:    // Attempt 1 takes the plan's window. Attempt k ≥ 2 is registered only after every earlier holdout is scored or
267:    // burned, and starts on the first whole UTC day after its registration (a requested start may be later, never
268:    // earlier), with 28 entry days and the plan's tail length.
269:    let aw: AttemptWindow;
270:    if (req.index === 1) aw = { fromDay: plan.fromDay, entryCutoffDay: plan.entryCutoffDay, tailEndDay: plan.tailEndDay };
271:    else {
272:      const open = s.attempts.flatMap((x) => x.holdoutIds).filter((id) => s.registry.entries.find((e) => e.holdoutId === id)?.burned !== true);
273:      if (open.length > 0) throw new RangeError(`holdout attempt ${req.index}: earlier holdouts are not scored yet (${open.join(', ')})`);
274:      const earliest = addDays(now.toISOString().slice(0, 10), 1);
275:      const fromDay = req.fromDay ?? earliest;
276:      if (fromDay < earliest) throw new RangeError(`holdout attempt ${req.index}: its window cannot start before ${earliest}, the first whole UTC day after registration`);
277:      const cutoff = addDays(fromDay, LATER_ATTEMPT_ENTRY_DAYS);
278:      aw = { fromDay, entryCutoffDay: cutoff, tailEndDay: addDays(cutoff, daysBetween(plan.entryCutoffDay, plan.tailEndDay)) };
279:    }
280:    const clashing = s.attempts.filter((x) => aw.fromDay < x.window.tailEndDay && x.window.fromDay < aw.tailEndDay).map((x) => x.index);
281:    if (clashing.length > 0) throw new RangeError(`holdout attempt ${req.index}: its window overlaps attempt ${clashing.join(', ')}`);
282:    const window = { fromDay: aw.fromDay, toDay: addDays(aw.tailEndDay, -1) };
283:    const clash = startedOverlapping(s, window, null);
284:    if (clash.length > 0) throw new RangeError(`holdout attempt ${req.index}: its window overlaps ${clash.join(', ')}, already run`);
285:    const registry = req.entries.reduce((r, e) => registerHoldout(r, { holdoutId: e.holdoutId, universe: e.universe, configId: e.configId, ...window }), s.registry);
286:    const attempt: HoldoutAttempt = {
287:      index: req.index, alpha: attemptAlpha(plan, req.index), window: aw, holdoutIds: req.entries.map((e) => e.holdoutId),
288:      configIds: Object.fromEntries(req.entries.map((e) => [e.holdoutId, e.configId])), registeredAt: now.toISOString(), started: null, ended: null,
289:    };
290:    const store = { ...s, registry, attempts: [...s.attempts, attempt] };
291:    return { store, value: store };
292:  });

339:/**
340: * Runs an authorised holdout and seals it. Before anything runs, every universe's holdout must be registered with this
341: * exact configuration id and window, unburned and never run (no earlier start record); otherwise the attempt is
342: * refused and logged. A start record is written before the run, so a run killed mid-way spends the window. A run that
343: * fails burns its holdouts ('run-failed'). Returns only the hash and counts.
344: */
345:export const runAndSealHoldout = (o: RunOptions & { readonly ledgerPath: string }, t: HoldoutTargets): SealedHoldout => {
346:  const configId = holdoutConfigId(o, t);
347:  t.vcs?.check();
348:  if (!existsSync(t.registryPath)) throw new RangeError(`no holdout registry at ${t.registryPath}: holdout not registered`);
349:  let store = readHoldoutStore(t.registryPath);
350:  const save = (message: string) => saveStore(t.registryPath, store, t.vcs, message);
351:  const log = (holdoutId: string, outcome: HoldoutRunRecord['outcome'], reason: string) => {
352:    store = { ...store, runs: [...store.runs, { holdoutId, outcome, configId, ledgerPath: o.ledgerPath, at: new Date().toISOString(), reason }] };
353:  };
354:  const burnAll = (why: string) => {
355:    let reg = store.registry;
356:    for (const id of Object.values(t.byUniverse)) if (reg.entries.some((e) => e.holdoutId === id)) reg = burnHoldout(reg, id, 'run-failed', why).registry;
357:    store = { ...store, registry: reg };
358:  };
359:  const refuse = (holdoutId: string, reason: string): never => {
360:    log(holdoutId, 'refused', reason);
361:    save(`Holdout ${holdoutId}: refused (${reason})`);
362:    throw new RangeError(`holdout ${holdoutId}: ${reason}`);
363:  };
364:  for (const [u, id] of Object.entries(t.byUniverse)) {
365:    const e = store.registry.entries.find((x) => x.holdoutId === id);
366:    if (e === undefined) refuse(id, 'not registered');
367:    else if (e.burned) refuse(id, `burned (${e.burnReason})`);
368:    else if (store.runs.some((r) => r.holdoutId === id && r.outcome !== 'refused')) {
369:      // A start without an end is a run that died: the window is spent.
370:      if (e.seal === 'registered') burnAll(`holdout ${id} has a start record and no result`);
371:      refuse(id, 'window already run');
372:    } else if (e.universe !== u) refuse(id, `registered for ${e.universe}, run for ${u}`);
373:    else if (e.configId !== configId) refuse(id, `configuration ${configId} is not the authorised ${e.configId}`);
374:    else if (e.fromDay !== t.window.fromDay || e.toDay !== t.window.toDay) refuse(id, `window ${t.window.fromDay}..${t.window.toDay} is not the authorised ${e.fromDay}..${e.toDay}`);
375:  }
376:  // One sealed ledger and one endpoint for every holdout of the attempt: the run must feed all of them.
377:  const ids = Object.values(t.byUniverse);
378:  const attempt = store.attempts.find((x) => ids.every((id) => x.holdoutIds.includes(id)));
379:  if (attempt === undefined || attempt.holdoutIds.length !== ids.length) refuse(ids[0] ?? '?', 'the run must feed every holdout of its registered attempt, and only those');
380:  const markAttempt = (patch: Partial<HoldoutAttempt>) => {
381:    store = { ...store, attempts: store.attempts.map((x) => (x.index === attempt!.index ? { ...x, ...patch } : x)) };
382:  };
383:  for (const id of ids) log(id, 'started', '');
384:  markAttempt({ started: new Date().toISOString() });
385:  save(`Holdout ${ids.join(', ')}: started`);
386:  t.vcs?.check();
387:  // Entries stop at the attempt's cutoff; the run keeps observing to the end of the tail so every position can finish.
388:  const cutoff = Date.parse(`${attempt!.window.entryCutoffDay}T00:00:00Z`);
389:  // Rows outside the authorised window stop the run (and so burn the holdout).
390:  const inWindow = (): Iterator<DatasetRow> => {
391:    const it = o.rows();
392:    return { next: () => {
393:      const r = it.next();
